\item{\bf A03T.3} For each state $s_i$, calculate the probability probability of reaching $s_2$ while avoiding $s_3$. Explain your solution in terms of the formal semantics based on cylinder sets and double-check your solution against PRISM. 
%
\ifwithanswers
\color{blue}
\par
We label the states with $AP=\{end,\ death\}$ and
\[
L(s_2)=\{end\},\qquad L(s_3)=\{death\},\qquad L(s_0)=L(s_1)=\emptyset .
\]
The property to check is $\phi=\lnot death\ \mathcal{U}\ end$. For an infinite path $\pi$, writing $\pi[k]$ for its $k$-th state, the semantics is
\[
\pi\vDash\phi \iff \exists k\in\mathbb{N}.\ \big(\pi[k]\vDash end\ \land\ \forall j<k.\ \pi[j]\vDash\lnot death\big).
\]
Let $Sat(\phi)=\{\pi\mid \pi\vDash\phi\}$. For each state $s_i$ we want $Pr_{s_i}(\phi)=Pr_{s_i}(Sat(\phi))$, where $Pr_{s_i}$ is the measure with $\iota(s_i)=1$ and
\[
Pr_{s_i}\big(Cyl(t_0t_1\dots t_n)\big)=\iota(t_0)\cdot\prod_{k=0}^{n-1}P(t_k,t_{k+1}).
\]
For every state we proceed in the same way: first we describe $Paths(s_i)$ with an exhaustive family of cylinder sets, then we check through the semantics which of them satisfy $\phi$, then we compute their probabilities, and finally we combine them into the result.

\medskip
\noindent\textbf{State $s_2$.}
\underline{Cylinder sets.} Every path from $s_2$ starts with $s_2$, so $Paths(s_2)=Cyl(s_2)$.
\underline{Semantics.} Taking $k=0$ we have $\pi[0]=s_2\vDash end$, and there is no $j<0$ to check, so the condition on $\lnot death$ holds vacuously. Hence $Cyl(s_2)\subseteq Sat(\phi)$.
\underline{Probability.} We get
\[
Pr_{s_2}(Cyl(s_2))=\iota(s_2)=1 .
\]
\underline{Result.} Therefore $Pr_{s_2}(\phi)=1$.

\medskip
\noindent\textbf{State $s_1$.}
\underline{Cylinder sets.} The only successors of $s_1$ are $s_1$ and $s_2$, so a path either stays in $s_1$ forever or leaves it towards $s_2$ after $n-1$ self-loops. This gives the disjoint and exhaustive decomposition
\[
Paths(s_1)=\{s_1^\omega\}\ \uplus\ \biguplus_{n\ge1}Cyl(s_1^ns_2).
\]
\underline{Semantics.} For a path in $Cyl(s_1^ns_2)$ we take $k=n$: then $\pi[n]=s_2\vDash end$ and $\pi[j]=s_1\vDash\lnot death$ for all $j<n$, so $Cyl(s_1^ns_2)\subseteq Sat(\phi)$. The path $s_1^\omega$ never visits a state satisfying $end$, so $s_1^\omega\notin Sat(\phi)$.
\underline{Probabilities.} The cylinder set $Cyl(s_1^ns_2)$ requires $n-1$ self-loops, each with probability $\tfrac12$, followed by the move to $s_2$, again with probability $\tfrac12$:
\[
Pr_{s_1}\big(Cyl(s_1^ns_2)\big)=\iota(s_1)\cdot\left(\tfrac12\right)^{n-1}\cdot\tfrac12=1\cdot\left(\tfrac12\right)^{n-1}\cdot\tfrac12=\left(\tfrac12\right)^{n}.
\]
The path that stays in $s_1$ forever has probability bounded by that of staying $n$ steps, for every $n$:
\[
Pr_{s_1}\big(\{s_1^\omega\}\big)=\lim_{n\to\infty}\iota(s_1)\cdot\left(\tfrac12\right)^{n}=\lim_{n\to\infty}1\cdot\left(\tfrac12\right)^{n}=0 .
\]
\underline{Result.} Only the cylinder sets ending in $s_2$ satisfy $\phi$, and the remaining path has measure $0$, so by $\sigma$-additivity and the geometric series
\[
Pr_{s_1}(\phi)=\sum_{n\ge1}\left(\tfrac12\right)^{n}=\frac{\tfrac12}{1-\tfrac12}=1 .
\]

\medskip
\noindent\textbf{State $s_3$.}
\underline{Cylinder sets.} Since $P(s_3,s_2)=1$, every path from $s_3$ starts with $s_3s_2$, hence $Paths(s_3)=Cyl(s_3s_2)$.
\underline{Semantics.} Here $\pi[0]=s_3\vDash death$ and $s_3\nvDash end$. The witness $k=0$ fails because $\pi[0]\nvDash end$, and any witness $k\ge1$ fails because it would require $\pi[0]\vDash\lnot death$. Hence $Cyl(s_3s_2)\cap Sat(\phi)=\emptyset$.
\underline{Probability.} We have
\[
Pr_{s_3}(Cyl(s_3s_2))=\iota(s_3)\cdot P(s_3,s_2)=1\cdot1=1,
\]
and this whole mass belongs to paths that violate $\phi$.
\underline{Result.} Therefore $Pr_{s_3}(\phi)=Pr_{s_3}(\emptyset)=0$.

\medskip
\noindent\textbf{State $s_0$.}
\underline{Cylinder sets.} The successors of $s_0$ are $s_3$ and $s_1$. Decomposing $Cyl(s_0s_1)$ as done for $s_1$, we get the disjoint and exhaustive family
\[
Paths(s_0)=Cyl(s_0s_3)\ \uplus\ \{s_0s_1^\omega\}\ \uplus\ \biguplus_{n\ge1}Cyl(s_0s_1^ns_2).
\]
\underline{Semantics.} In $Cyl(s_0s_3)$ we have $\pi[0]=s_0\nvDash end$, so $k=0$ fails; also $\pi[1]=s_3\nvDash end$, so $k=1$ fails, and any $k\ge2$ fails because $\pi[1]=s_3\vDash death$. Hence $Cyl(s_0s_3)\cap Sat(\phi)=\emptyset$. In $Cyl(s_0s_1^ns_2)$ we take $k=n+1$: then $\pi[n+1]=s_2\vDash end$ and $\pi[j]\in\{s_0,s_1\}\vDash\lnot death$ for all $j\le n$, so $Cyl(s_0s_1^ns_2)\subseteq Sat(\phi)$. Finally $s_0s_1^\omega$ never reaches $s_2$, so $s_0s_1^\omega\notin Sat(\phi)$.
\underline{Probabilities.} We compute
\begin{align*}
Pr_{s_0}(Cyl(s_0s_3))&=\iota(s_0)\cdot P(s_0,s_3)=1\cdot0.6=0.6,\\
Pr_{s_0}(Cyl(s_0s_1^ns_2))&=1\cdot0.4\cdot\left(\tfrac12\right)^{n-1}\cdot\tfrac12=0.4\cdot\left(\tfrac12\right)^{n},\\
Pr_{s_0}(\{s_0s_1^\omega\})&=\lim_{n\to\infty}1\cdot0.4\cdot\left(\tfrac12\right)^{n-1}=0 .
\end{align*}
\underline{Result.} Only the cylinder sets $Cyl(s_0s_1^ns_2)$ satisfy $\phi$, so
\[
Pr_{s_0}(\phi)=\sum_{n\ge1}0.4\cdot\left(\tfrac12\right)^{n}=0.4\cdot\frac{\tfrac12}{1-\tfrac12}=0.4 .
\]

\medskip
\noindent In summary,
\[
Pr_{s_0}(\phi)=0.4,\qquad Pr_{s_1}(\phi)=1,\qquad Pr_{s_2}(\phi)=1,\qquad Pr_{s_3}(\phi)=0 .
\]
